# Program to print day of the week based on a number

# Prompt the user to enter a number between 1 and 7
day_num = int(input("Enter a number (1-7): "))

# Check the number and print the corresponding day
if day_num == 1:
    print("Monday")
elif day_num == 2:
    print("Tuesday")
elif day_num == 3:
    print("Wednesday")
elif day_num == 4:
    print("Thursday")
elif day_num == 5:
    print("Friday")
elif day_num == 6:
    print("Saturday")
elif day_num == 7:
    print("Sunday")
else:
    # Handle invalid numbers
    print("Invalid number, please enter a number between 1 and 7")
