# Program to check whether a number is even or odd

# Prompt the user to enter a number
num = int(input("Enter a number: "))  # Convert input to integer

# Use modulus operator (%) to find remainder when divided by 2
# If remainder is 0, number is even; otherwise, it is odd
if num % 2 == 0:
    print("The number is even")  # Prints if the number is divisible by 2
else:
    print("The number is odd")   # Prints if the number is not divisible by 2
